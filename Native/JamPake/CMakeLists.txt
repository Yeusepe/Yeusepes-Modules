cmake_minimum_required(VERSION 3.22)
project(JamPake LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(OPENSSL_NO_ASM ON CACHE BOOL "" FORCE)
if(NOT DEFINED BORINGSSL_SOURCE)
  message(FATAL_ERROR "Pass -DBORINGSSL_SOURCE=<pinned checkout>; see build.ps1")
endif()
add_subdirectory("${BORINGSSL_SOURCE}" boringssl EXCLUDE_FROM_ALL)
add_library(JamPake SHARED jam_pake.cc)
target_link_libraries(JamPake PRIVATE crypto)
target_include_directories(JamPake PRIVATE "${BORINGSSL_SOURCE}/include")
target_compile_definitions(JamPake PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
